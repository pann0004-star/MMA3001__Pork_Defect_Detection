blocked_split

Create a leakage resistant train/valid/test split from blocks of video frames.

Roboflow split the frames of one video at random, so neighbouring (near identical) frames ended up in different splits. This script regroups the same images so that whole stretches of video go to a single split, then removes frames near the boundaries so no valid or test frame has a close neighbour in train.

The original data folder is left untouched, so both splits can be trained and compared.

Run from the repo root:

python src/blocked_split.py --data data --out data/blocked

 40@dataclass
 41class Frame:
 42    """Every exported copy of one video frame.
 43
 44    Attributes:
 45        number: Video frame number.
 46        copies: One entry per exported image of this frame. Train frames
 47            usually have several (augmented) copies.
 48        class_counts: Boxes per class id, counted from a single copy.
 49    """
 50
 51    number: int
 52    copies: list[Copy] = field(default_factory=list)
 53    class_counts: np.ndarray | None = None
 54
 55    def is_ambiguous(self, aug_copies: int = 3) -> bool:
 56        """Return True if this frame number probably covers more than one photo.
 57
 58        Roboflow gave each training photo ``aug_copies`` exported versions, so
 59        more copies than that, or copies spread over several original splits,
 60        means the number was reused (e.g. by a second recording).
 61
 62        Args:
 63            aug_copies: Exported versions per training photo.
 64        """
 65        train_copies = sum(1 for c in self.copies if c[0] == "train")
 66        return train_copies > aug_copies or len({c[0] for c in self.copies}) > 1
 67
 68    def clean_copy(self) -> Copy:
 69        """Return one copy, preferring an unaugmented one from test or valid.
 70
 71        Roboflow only augmented the training split, so a copy exported to
 72        test or valid is the closest thing to the untouched original.
 73        """
 74        for preferred in ("test", "valid", "train"):
 75            for copy in self.copies:
 76                if copy[0] == preferred:
 77                    return copy
 78        return self.copies[0]

 81def collect_frames(data_dir: Path, n_classes: int) -> tuple[dict[int, Frame], list[str]]:
 82    """Group every image in the dataset by its video frame number.
 83
 84    Args:
 85        data_dir: Roboflow dataset root with train, valid and test folders.
 86        n_classes: Number of classes in the dataset.
 87
 88    Returns:
 89        A tuple of (frames keyed by frame number, names of images skipped
 90        because they contain no frame number).
 91    """
 92    frames: dict[int, Frame] = {}
 93    skipped: list[str] = []
 94    for split in SPLITS:
 95        img_dir = Path(data_dir) / split / "images"
 96        lbl_dir = Path(data_dir) / split / "labels"
 97        for img in sorted(img_dir.iterdir()):
 98            if img.suffix.lower() not in IMAGE_EXTS:
 99                continue
100            number = frame_number(img.name)
101            if number is None:
102                skipped.append(img.name)
103                continue
104            lbl = lbl_dir / f"{img.stem}.txt"
105            frames.setdefault(number, Frame(number)).copies.append(
106                (split, img, lbl if lbl.exists() else None)
107            )
108
109    # Augmentation only changed colour and exposure, so every copy has the
110    # same boxes. Count them once to avoid inflating the totals.
111    for frame in frames.values():
112        counts = np.zeros(n_classes, dtype=int)
113        label = frame.clean_copy()[2]
114        if label is not None:
115            boxes, _ = parse_label_file(label, n_classes)
116            for box in boxes:
117                counts[box[0]] += 1
118        frame.class_counts = counts
119    return frames, skipped

122def make_blocks(numbers: np.ndarray, block_size: int) -> np.ndarray:
123    """Assign each frame to a block of consecutive frame numbers.
124
125    Args:
126        numbers: Frame numbers.
127        block_size: Width of each block in frames (e.g. 300 means frames
128            0 to 299 form block 0, 300 to 599 block 1, ...).
129
130    Returns:
131        Block index per frame, renumbered 0, 1, 2... with no gaps.
132    """
133    _, block_ids = np.unique(np.asarray(numbers) // block_size, return_inverse=True)
134    return block_ids

137def score_assignment(
138    assign: np.ndarray, block_frames: np.ndarray, block_classes: np.ndarray, targets: np.ndarray
139) -> float:
140    """Score how well a block to split assignment meets the targets (lower is better).
141
142    The score adds three terms:
143
144    1. how far each split's share of frames is from its target fraction;
145    2. how far each class's share in each split is from that target;
146    3. a penalty for every class missing from a split: 100 if missing from
147       train (the model cannot learn it at all), 10 if missing from valid
148       or test (it cannot be evaluated). So when a rare class is too
149       clustered to share, train always keeps it.
150
151    Args:
152        assign: Split index (0 train, 1 valid, 2 test) for each block.
153        block_frames: Number of frames in each block.
154        block_classes: Box counts per class for each block, shape (blocks, classes).
155        targets: Target fraction of frames for each split, summing to 1.
156
157    Returns:
158        The combined score.
159    """
160    n_splits = len(targets)
161    frames = np.bincount(assign, weights=block_frames, minlength=n_splits)
162    size_err = np.abs(frames / frames.sum() - targets).sum()
163
164    per_split = np.zeros((n_splits, block_classes.shape[1]))
165    np.add.at(per_split, assign, block_classes)
166    totals = per_split.sum(axis=0)
167    present = totals > 0
168    shares = per_split[:, present] / totals[present]
169    balance_err = np.abs(shares - targets[:, None]).mean(axis=1).sum()
170
171    missing = (per_split[:, present] == 0).sum(axis=1)  # per split: train, valid, test
172    penalty = 100 * missing[0] + 10 * missing[1:].sum()
173    return float(size_err + balance_err + penalty)

176def search_assignment(
177    block_frames: np.ndarray,
178    block_classes: np.ndarray,
179    targets: np.ndarray,
180    n_trials: int = 20000,
181    seed: int = 0,
182) -> tuple[np.ndarray, float]:
183    """Find a good block assignment by random search.
184
185    Each trial sends every block to a random split (weighted by the targets)
186    and keeps the assignment with the lowest score. A fixed seed makes the
187    result reproducible.
188
189    Args:
190        block_frames: Number of frames in each block.
191        block_classes: Box counts per class for each block.
192        targets: Target fraction of frames for each split.
193        n_trials: Number of random assignments to try.
194        seed: Random seed.
195
196    Returns:
197        A tuple of (best assignment, its score).
198    """
199    rng = np.random.default_rng(seed)
200    best, best_score = None, np.inf
201    for _ in range(n_trials):
202        assign = rng.choice(len(targets), size=len(block_frames), p=targets)
203        score = score_assignment(assign, block_frames, block_classes, targets)
204        if score < best_score:
205            best, best_score = assign, score
206    return best, best_score

209def purge_boundaries(
210    numbers: np.ndarray, split_ids: np.ndarray, gap: int, protected: np.ndarray | None = None
211) -> np.ndarray:
212    """Mark frames to drop because they sit too close to another split.
213
214    Train frames within ``gap`` frames of any valid or test frame are
215    dropped, as are valid frames within ``gap`` of a test frame. Protected
216    train frames are never dropped; instead, any valid or test frame within
217    ``gap`` of them is dropped.
218
219    Args:
220        numbers: Frame numbers.
221        split_ids: Split index per frame (0 train, 1 valid, 2 test).
222        gap: Minimum allowed frame distance between different splits.
223        protected: Optional boolean mask of train frames that must be kept.
224
225    Returns:
226        Boolean mask, True for frames to keep.
227    """
228    numbers = np.asarray(numbers)
229    split_ids = np.asarray(split_ids)
230    protected = np.zeros(len(numbers), dtype=bool) if protected is None else np.asarray(protected)
231    keep = np.ones(len(numbers), dtype=bool)
232
233    # Clear evaluation frames next to protected training frames first.
234    evaluation = split_ids > 0
235    if protected.any() and evaluation.any():
236        dist = nearest_frame_distance(numbers[evaluation], numbers[protected])
237        keep[np.flatnonzero(evaluation)[dist <= gap]] = False
238
239    for split in range(len(SPLITS) - 1):
240        mine = (split_ids == split) & keep & ~protected
241        later = numbers[(split_ids > split) & keep]
242        dist = nearest_frame_distance(numbers[mine], later)
243        keep[np.flatnonzero(mine)[dist <= gap]] = False
244    return keep

247def write_dataset(
248    out_dir: Path, by_split: dict[str, list[Frame]], names: list[str], blocks: dict[int, int]
249) -> dict[str, int]:
250    """Copy images and labels into the new split folders.
251
252    Train keeps every augmented copy of its frames. Valid and test get one
253    clean copy per frame, so evaluation never counts the same frame twice.
254
255    Args:
256        out_dir: Destination folder (created if needed).
257        by_split: Frames assigned to each split.
258        names: Class names, written to the new data.yaml.
259        blocks: Block index per frame number, recorded in the manifest.
260
261    Returns:
262        Number of images written per split.
263    """
264    written = {}
265    rows = []
266    for split, frames in by_split.items():
267        img_dir = out_dir / split / "images"
268        lbl_dir = out_dir / split / "labels"
269        img_dir.mkdir(parents=True, exist_ok=True)
270        lbl_dir.mkdir(parents=True, exist_ok=True)
271        count = 0
272        for frame in frames:
273            copies = frame.copies if split == "train" else [frame.clean_copy()]
274            for _, img, lbl in copies:
275                shutil.copy2(img, img_dir / img.name)
276                if lbl is not None:
277                    shutil.copy2(lbl, lbl_dir / lbl.name)
278                count += 1
279            rows.append([frame.number, blocks[frame.number], split, len(copies),
280                         *frame.class_counts.tolist()])
281        written[split] = count
282
283    cfg = {"train": "train/images", "val": "valid/images", "test": "test/images",
284           "nc": len(names), "names": names}
285    with open(out_dir / "data.yaml", "w", encoding="utf-8") as f:
286        yaml.safe_dump(cfg, f, sort_keys=False)
287
288    with open(out_dir / "split_manifest.csv", "w", newline="", encoding="utf-8") as f:
289        writer = csv.writer(f)
290        writer.writerow(["frame", "block", "split", "images_written", *names])
291        writer.writerows(sorted(rows))
292    return written

295def blocked_split(
296    data_dir: Path,
297    out_dir: Path,
298    block_size: int = 300,
299    gap: int = 25,
300    targets: tuple[float, float, float] = (0.7, 0.15, 0.15),
301    n_trials: int = 20000,
302    seed: int = 0,
303    overwrite: bool = False,
304) -> dict:
305    """Run the full blocked split and write the new dataset.
306
307    Args:
308        data_dir: Original Roboflow dataset root.
309        out_dir: Where to write the blocked dataset.
310        block_size: Width of each block in frames.
311        gap: Minimum frame distance allowed between different splits.
312        targets: Target fraction of frames for train, valid and test.
313        n_trials: Random search trials.
314        seed: Random seed for reproducibility.
315        overwrite: Replace ``out_dir`` if it already exists.
316
317    Returns:
318        Summary dictionary used for the console report.
319
320    Raises:
321        ValueError: If targets do not sum to 1 or ``out_dir`` is ``data_dir``.
322        FileExistsError: If ``out_dir`` exists and ``overwrite`` is False.
323    """
324    data_dir, out_dir = Path(data_dir), Path(out_dir)
325    targets_arr = np.asarray(targets, dtype=float)
326    if not np.isclose(targets_arr.sum(), 1.0):
327        raise ValueError("targets must sum to 1")
328    if out_dir.resolve() == data_dir.resolve():
329        raise ValueError("out_dir must differ from data_dir, the original data is kept")
330    if out_dir.exists():
331        if not overwrite:
332            raise FileExistsError(f"{out_dir} exists, use --overwrite to replace it")
333        shutil.rmtree(out_dir)
334
335    names = load_class_names(data_dir)
336    frames, skipped = collect_frames(data_dir, len(names))
337    numbers = np.array(sorted(frames))
338    class_matrix = np.array([frames[n].class_counts for n in numbers])
339
340    block_ids = make_blocks(numbers, block_size)
341    n_blocks = block_ids.max() + 1
342    block_frames = np.bincount(block_ids, minlength=n_blocks)
343    block_classes = np.zeros((n_blocks, len(names)))
344    np.add.at(block_classes, block_ids, class_matrix)
345
346    assign, score = search_assignment(block_frames, block_classes, targets_arr, n_trials, seed)
347    split_ids = assign[block_ids].copy()
348
349    # Frame numbers shared by two photos always go to train, so valid and test
350    # only contain frames that are unambiguously one photo.
351    ambiguous = np.array([frames[n].is_ambiguous() for n in numbers])
352    split_ids[ambiguous] = 0
353    keep = purge_boundaries(numbers, split_ids, gap, protected=ambiguous)
354
355    by_split = {s: [frames[n] for n in numbers[(split_ids == i) & keep]]
356                for i, s in enumerate(SPLITS)}
357    written = write_dataset(out_dir, by_split, names,
358                            dict(zip(numbers.tolist(), block_ids.tolist())))
359
360    kept_numbers = {s: np.array([f.number for f in fs]) for s, fs in by_split.items()}
361    return {
362        "names": names,
363        "n_frames": len(numbers),
364        "n_blocks": int(n_blocks),
365        "score": score,
366        "skipped": skipped,
367        "ambiguous": int(ambiguous.sum()),
368        "purged": {s: int(((split_ids == i) & ~keep).sum()) for i, s in enumerate(SPLITS)},
369        "frames": {s: len(fs) for s, fs in by_split.items()},
370        "images": written,
371        "classes": {s: (np.sum([f.class_counts for f in fs], axis=0) if fs
372                        else np.zeros(len(names), dtype=int))
373                    for s, fs in by_split.items()},
374        "min_gap": {
375            "train > valid": _min_gap(kept_numbers["valid"], kept_numbers["train"]),
376            "train > test": _min_gap(kept_numbers["test"], kept_numbers["train"]),
377            "valid > test": _min_gap(kept_numbers["test"], kept_numbers["valid"]),
378        },
379    }

382def _min_gap(query: np.ndarray, reference: np.ndarray) -> float:
383    """Smallest frame distance from any query frame to the reference (inf if either is empty)."""
384    if len(query) == 0 or len(reference) == 0:
385        return float("inf")
386    return float(nearest_frame_distance(query, reference).min())

389def print_summary(summary: dict, gap: int) -> None:
390    """Print the outcome of the blocked split."""
391    print(f"\n{summary['n_frames']} unique frames in {summary['n_blocks']} blocks, "
392          f"search score {summary['score']:.3f}")
393    print(f"{summary['ambiguous']} frame numbers shared by two photos, kept in train")
394    if summary["skipped"]:
395        print(f"Skipped {len(summary['skipped'])} images with no frame number")
396
397    total = sum(summary["frames"].values())
398    print("\n== New split ==")
399    print(f"{'split':<6} {'frames':>7} {'share':>7} {'images':>7} {'purged':>7}")
400    for s in SPLITS:
401        print(f"{s:<6} {summary['frames'][s]:>7} {100 * summary['frames'][s] / total:>6.1f}% "
402              f"{summary['images'][s]:>7} {summary['purged'][s]:>7}")
403
404    print("\n== Boxes per class (one copy per frame) ==")
405    print(f"{'class':<16}" + "".join(f"{s:>8}" for s in SPLITS))
406    missing = []
407    for cid, name in enumerate(summary["names"]):
408        counts = [int(summary["classes"][s][cid]) for s in SPLITS]
409        print(f"{name:<16}" + "".join(f"{c:>8}" for c in counts))
410        missing += [f"{name} in {s}" for s, c in zip(SPLITS, counts) if c == 0]
411
412    print(f"\n== Closest frames between splits (gap setting {gap}) ==")
413    for pair, d in summary["min_gap"].items():
414        print(f"{pair:<14} {d:>6.0f}  {'OK' if d > gap else 'TOO CLOSE'}")
415
416    if missing:
417        print("\nWARNING, classes with no boxes: " + ", ".join(missing))

420def main(argv: list[str] | None = None) -> int:
421    """Command line entry point.
422
423    Args:
424        argv: Argument list (defaults to ``sys.argv``).
425
426    Returns:
427        Process exit code.
428    """
429    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
430    parser.add_argument("--data", default="data", help="original dataset root")
431    parser.add_argument("--out", default="data/blocked", help="output folder")
432    parser.add_argument("--block-size", type=int, default=300, help="frames per block")
433    parser.add_argument("--gap", type=int, default=25, help="min frames between splits")
434    parser.add_argument("--targets", type=float, nargs=3, default=[0.7, 0.15, 0.15],
435                        metavar=("TRAIN", "VALID", "TEST"), help="target frame fractions")
436    parser.add_argument("--trials", type=int, default=20000, help="random search trials")
437    parser.add_argument("--seed", type=int, default=0, help="random seed")
438    parser.add_argument("--overwrite", action="store_true", help="replace existing output")
439    args = parser.parse_args(argv)
440
441    summary = blocked_split(args.data, args.out, args.block_size, args.gap,
442                            tuple(args.targets), args.trials, args.seed, args.overwrite)
443    print_summary(summary, args.gap)
444    print(f"\nWritten to {args.out}")
445    return 0

def score_assignment( assign: numpy.ndarray, block_frames: numpy.ndarray, block_classes: numpy.ndarray, targets: numpy.ndarray) -> float: 
138def score_assignment(
139    assign: np.ndarray, block_frames: np.ndarray, block_classes: np.ndarray, targets: np.ndarray
140) -> float:
141    """Score how well a block to split assignment meets the targets (lower is better).
142
143    The score adds three terms:
144
145    1. how far each split's share of frames is from its target fraction;
146    2. how far each class's share in each split is from that target;
147    3. a penalty for every class missing from a split: 100 if missing from
148       train (the model cannot learn it at all), 10 if missing from valid
149       or test (it cannot be evaluated). So when a rare class is too
150       clustered to share, train always keeps it.
151
152    Args:
153        assign: Split index (0 train, 1 valid, 2 test) for each block.
154        block_frames: Number of frames in each block.
155        block_classes: Box counts per class for each block, shape (blocks, classes).
156        targets: Target fraction of frames for each split, summing to 1.
157
158    Returns:
159        The combined score.
160    """
161    n_splits = len(targets)
162    frames = np.bincount(assign, weights=block_frames, minlength=n_splits)
163    size_err = np.abs(frames / frames.sum() - targets).sum()
164
165    per_split = np.zeros((n_splits, block_classes.shape[1]))
166    np.add.at(per_split, assign, block_classes)
167    totals = per_split.sum(axis=0)
168    present = totals > 0
169    shares = per_split[:, present] / totals[present]
170    balance_err = np.abs(shares - targets[:, None]).mean(axis=1).sum()
171
172    missing = (per_split[:, present] == 0).sum(axis=1)  # per split: train, valid, test
173    penalty = 100 * missing[0] + 10 * missing[1:].sum()
174    return float(size_err + balance_err + penalty)

def search_assignment( block_frames: numpy.ndarray, block_classes: numpy.ndarray, targets: numpy.ndarray, n_trials: int = 20000, seed: int = 0) -> tuple[numpy.ndarray, float]: 
177def search_assignment(
178    block_frames: np.ndarray,
179    block_classes: np.ndarray,
180    targets: np.ndarray,
181    n_trials: int = 20000,
182    seed: int = 0,
183) -> tuple[np.ndarray, float]:
184    """Find a good block assignment by random search.
185
186    Each trial sends every block to a random split (weighted by the targets)
187    and keeps the assignment with the lowest score. A fixed seed makes the
188    result reproducible.
189
190    Args:
191        block_frames: Number of frames in each block.
192        block_classes: Box counts per class for each block.
193        targets: Target fraction of frames for each split.
194        n_trials: Number of random assignments to try.
195        seed: Random seed.
196
197    Returns:
198        A tuple of (best assignment, its score).
199    """
200    rng = np.random.default_rng(seed)
201    best, best_score = None, np.inf
202    for _ in range(n_trials):
203        assign = rng.choice(len(targets), size=len(block_frames), p=targets)
204        score = score_assignment(assign, block_frames, block_classes, targets)
205        if score < best_score:
206            best, best_score = assign, score
207    return best, best_score

def purge_boundaries( numbers: numpy.ndarray, split_ids: numpy.ndarray, gap: int, protected: numpy.ndarray | None = None) -> numpy.ndarray: 
210def purge_boundaries(
211    numbers: np.ndarray, split_ids: np.ndarray, gap: int, protected: np.ndarray | None = None
212) -> np.ndarray:
213    """Mark frames to drop because they sit too close to another split.
214
215    Train frames within ``gap`` frames of any valid or test frame are
216    dropped, as are valid frames within ``gap`` of a test frame. Protected
217    train frames are never dropped; instead, any valid or test frame within
218    ``gap`` of them is dropped.
219
220    Args:
221        numbers: Frame numbers.
222        split_ids: Split index per frame (0 train, 1 valid, 2 test).
223        gap: Minimum allowed frame distance between different splits.
224        protected: Optional boolean mask of train frames that must be kept.
225
226    Returns:
227        Boolean mask, True for frames to keep.
228    """
229    numbers = np.asarray(numbers)
230    split_ids = np.asarray(split_ids)
231    protected = np.zeros(len(numbers), dtype=bool) if protected is None else np.asarray(protected)
232    keep = np.ones(len(numbers), dtype=bool)
233
234    # Clear evaluation frames next to protected training frames first.
235    evaluation = split_ids > 0
236    if protected.any() and evaluation.any():
237        dist = nearest_frame_distance(numbers[evaluation], numbers[protected])
238        keep[np.flatnonzero(evaluation)[dist <= gap]] = False
239
240    for split in range(len(SPLITS) - 1):
241        mine = (split_ids == split) & keep & ~protected
242        later = numbers[(split_ids > split) & keep]
243        dist = nearest_frame_distance(numbers[mine], later)
244        keep[np.flatnonzero(mine)[dist <= gap]] = False
245    return keep

250) -> dict[str, int]:

252

255
256    Args:
257        out_dir: Destination folder (created if needed).

def blocked_split( data_dir: pathlib.Path, out_dir: pathlib.Path, block_size: int = 300, gap: int = 25, targets: tuple[float, float, float] = (0.7, 0.15, 0.15), n_trials: int = 20000, seed: int = 0, overwrite: bool = False) -> dict: 
296def blocked_split(
297    data_dir: Path,
298    out_dir: Path,
299    block_size: int = 300,
300    gap: int = 25,
301    targets: tuple[float, float, float] = (0.7, 0.15, 0.15),
302    n_trials: int = 20000,
303    seed: int = 0,
304    overwrite: bool = False,
305) -> dict:
306    """Run the full blocked split and write the new dataset.
307
308    Args:
309        data_dir: Original Roboflow dataset root.
310        out_dir: Where to write the blocked dataset.
311        block_size: Width of each block in frames.
312        gap: Minimum frame distance allowed between different splits.
313        targets: Target fraction of frames for train, valid and test.
314        n_trials: Random search trials.
315        seed: Random seed for reproducibility.
316        overwrite: Replace ``out_dir`` if it already exists.
317
318    Returns:
319        Summary dictionary used for the console report.
320
321    Raises:
322        ValueError: If targets do not sum to 1 or ``out_dir`` is ``data_dir``.
323        FileExistsError: If ``out_dir`` exists and ``overwrite`` is False.
324    """
325    data_dir, out_dir = Path(data_dir), Path(out_dir)
326    targets_arr = np.asarray(targets, dtype=float)
327    if not np.isclose(targets_arr.sum(), 1.0):
328        raise ValueError("targets must sum to 1")
329    if out_dir.resolve() == data_dir.resolve():
330        raise ValueError("out_dir must differ from data_dir, the original data is kept")
331    if out_dir.exists():
332        if not overwrite:
333            raise FileExistsError(f"{out_dir} exists, use --overwrite to replace it")
334        shutil.rmtree(out_dir)
335
336    names = load_class_names(data_dir)
337    frames, skipped = collect_frames(data_dir, len(names))
338    numbers = np.array(sorted(frames))
339    class_matrix = np.array([frames[n].class_counts for n in numbers])
340
341    block_ids = make_blocks(numbers, block_size)
342    n_blocks = block_ids.max() + 1
343    block_frames = np.bincount(block_ids, minlength=n_blocks)
344    block_classes = np.zeros((n_blocks, len(names)))
345    np.add.at(block_classes, block_ids, class_matrix)
346
347    assign, score = search_assignment(block_frames, block_classes, targets_arr, n_trials, seed)
348    split_ids = assign[block_ids].copy()
349
350    # Frame numbers shared by two photos always go to train, so valid and test
351    # only contain frames that are unambiguously one photo.
352    ambiguous = np.array([frames[n].is_ambiguous() for n in numbers])
353    split_ids[ambiguous] = 0
354    keep = purge_boundaries(numbers, split_ids, gap, protected=ambiguous)
355
356    by_split = {s: [frames[n] for n in numbers[(split_ids == i) & keep]]
357                for i, s in enumerate(SPLITS)}
358    written = write_dataset(out_dir, by_split, names,
359                            dict(zip(numbers.tolist(), block_ids.tolist())))
360
361    kept_numbers = {s: np.array([f.number for f in fs]) for s, fs in by_split.items()}
362    return {
363        "names": names,
364        "n_frames": len(numbers),
365        "n_blocks": int(n_blocks),
366        "score": score,
367        "skipped": skipped,
368        "ambiguous": int(ambiguous.sum()),
369        "purged": {s: int(((split_ids == i) & ~keep).sum()) for i, s in enumerate(SPLITS)},
370        "frames": {s: len(fs) for s, fs in by_split.items()},
371        "images": written,
372        "classes": {s: (np.sum([f.class_counts for f in fs], axis=0) if fs
373                        else np.zeros(len(names), dtype=int))
374                    for s, fs in by_split.items()},
375        "min_gap": {
376            "train > valid": _min_gap(kept_numbers["valid"], kept_numbers["train"]),
377            "train > test": _min_gap(kept_numbers["test"], kept_numbers["train"]),
378            "valid > test": _min_gap(kept_numbers["test"], kept_numbers["valid"]),
379        },
380    }
